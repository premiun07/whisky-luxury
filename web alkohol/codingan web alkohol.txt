/* SIMPAN */
function simpan(){
localStorage.setItem("cart", JSON.stringify(cart));
}

/* TAMBAH */
function tambah(n){
if(!cart[n]) cart[n]=0;
cart[n]++;
animasi();
update();
}

/* KURANG (FIX BUG DI SINI) */
function kurang(n){
if(cart[n]){
cart[n]--;

if(cart[n] <= 0){
delete cart[n];
let el=document.getElementById("qty-"+n);
if(el) el.innerText = 0; // 🔥 FIX: tampilkan 0
}else{
let el=document.getElementById("qty-"+n);
if(el) el.innerText = cart[n];
}
}

update();
}

/* UPDATE (FIX TOTAL & UI) */
function update(){
let total=0, totalHarga=0;

for(let i in cart){
total += cart[i];
totalHarga += cart[i]*harga[i];
}

/* UPDATE SEMUA QTY */
dataProduk.forEach(p=>{
let nama = p[0];
let el = document.getElementById("qty-"+nama);
if(el){
el.innerText = cart[nama] ? cart[nama] : 0;
}
});

/* UPDATE CHECKOUT */
let checkout = document.querySelector(".checkout");
checkout.innerHTML = `🛒 <span class="badge">${total}</span> Rp ${totalHarga.toLocaleString()}`;

simpan();
}

/* ANIMASI */
function animasi(){
let el=document.createElement("div");
el.innerHTML="🛒";
el.className="fly";
el.style.left="50%";
el.style.top="50%";

document.body.appendChild(el);

setTimeout(()=>{
el.style.left="90%";
el.style.top="90%";
el.style.opacity=0;
},50);

setTimeout(()=>el.remove(),800);
}

}

pesan+=`%0ATotal: Rp ${total.toLocaleString()}`;
window.open("https://wa.me/6287756808208?text="+pesan);
}

/* LOAD AWAL */
update();
